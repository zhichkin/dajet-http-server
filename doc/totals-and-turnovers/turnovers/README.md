# Таблица итогов регистра накопления оборотов

В отличие от регистров накопления остатков, регистры накопления оборотов не имеют актуальных итогов, а также установленного периода рассчитанных итогов. Период рассчитанных итогов для обротных регистров устанавливается платформой автоматически при включении использования итогов. Минимальный и максимальный периоды хранимых итогов вычисляются по основной таблице движений регистра запросами следующего вида:

```SQL
SELECT MIN(Период) FROM РегистрНакопления.ПродажиТовара -- Минимальный период
SELECT MAX(Период) FROM РегистрНакопления.ПродажиТовара -- Максимальный период
```

Полученные таким образом значения, приводятся к началу месяца. Например, если основная таблица движений содержит записи с 15 января 2026 года по 15 мая 2026 года, то минимальный период хранимых итогов будет равен ```2026-01-01 00:00:00```, а максимальный - ```2026-05-01 00:00:00```.

Пересчёт итогов выполняется для каждого месяца, начиная с минимального периода хранимых итогов и заканчивая максимальным. Запрос пересчёта итогов выполняется для каждого месяца в отдельной транзакции СУБД, имеющей уровень изоляции READ COMMITTED, и выглядит следующим образом:

```SQL
-- Пересчёт итогов регистра накопления оборотов за январь месяц

DECLARE @НачалоПериода datetime = '2026-01-01 00:00:00'
DECLARE @КонецПериода  datetime = '2026-02-01 00:00:00'

DELETE РегистрНакполения.ПродажиТовара.Итоги WHERE Период = @НачалоПериода

INSERT РегистрНакполения.ПродажиТовара.Итоги (Период, Номенклатура, Количество, Splitter)
SELECT @НачалоПериода, Номенклатура, SUM(Количество), 0
  FROM РегистрНакполения.ПродажиТовара
 WHERE Активность = TRUE AND Период >= @НачалоПериода AND Период < @КонецПериода
 GROUP BY Номенклатура
```

> **Вывод:** обороты за месяц хранятся на начало этого месяца. 

Рассмотрим запросы, которые платформа генерирует для получения оборотов.

**Когда используется только таблица итогов ?**

Самый оптимальный запрос: начало периода отчёта совпадает с началом месяца, а конец периода отчёта с концом месяца.

```SQL
DECLARE @НачалоПериода datetime = '2026-03-01 00:00:00' -- Начало периода отчёта
DECLARE @КонецПериода  datetime = '2026-04-30 23:59:59' -- Конец периода отчёта

-- Платформа приводит конец периода отчёта к началу следующего месяца
SET @КонецПериода = DATEADD('SECOND', @КонецПериода)

SELECT Номенклатура, SUM(Количество) AS КоличествоОборот
  FROM РегистрНакполения.ПродажиТовара.Итоги
 WHERE Период >= @НачалоПериода AND Период < @КонецПериода
 GROUP BY Номенклатура
HAVING SUM(Количество) <> 0.0
```

Ниже следующий скрипт демонстрирует различные варианты построения отчёта для получения оборотов в зависимости от переданных пользователем параметров ```Начало периода``` и ```Конец периода```.

> **Важно!** Логика построения отчётов основана на использовании вирутальной таблицы "Обороты" и трассировке запросов СУБД.

## Получение оборотов регистра накопления при помощи DaJet Script

```SQL
DECLARE @НачалоПериода datetime = '2026-03-01T00:00:00'
DECLARE @КонецПериода  datetime = '2026-03-31T23:59:59'

PRIVATE @НачалоИтогов  datetime
PRIVATE @КонецИтогов   datetime
PRIVATE @ВариантОтчёта integer
PRIVATE @Настройки     object
PRIVATE @Результат     array

USE 'MS_TEST'

  SELECT UseTotals -- numeric(1,0) Использовать итоги
    INTO @Настройки
    FROM РегистрНакопления.ПродажиТовара.Настройки

  IF @Настройки.UseTotals = 1 THEN
    THROW 'Использование итогов выключено'
  END

  IF @НачалоПериода = DATESTART('MONTH', @НачалоПериода) THEN
    SET @НачалоИтогов = @НачалоПериода
  ELSE
    SET @НачалоИтогов = DATEADD('MONTH', 1, DATESTART('MONTH', @НачалоПериода))
  END

  IF @КонецПериода = DATEEND('MONTH', @КонецПериода) THEN
    SET @КонецИтогов  = DATEADD('MONTH', 1, DATESTART('MONTH', @КонецПериода))
  ELSE
    SET @КонецИтогов = DATESTART('MONTH', @КонецПериода)
  END

  IF @НачалоПериода = DATESTART('MONTH', @НачалоПериода) AND @КонецПериода = DATEEND('MONTH', @КонецПериода) THEN

    SET @ВариантОтчёта = 1 PRINT 'Начало и конец месяца - только итоги'
  
  ELSE IF DATESTART('MONTH', @НачалоПериода) = DATESTART('MONTH', @КонецПериода) THEN

         SET @ВариантОтчёта = 2 PRINT 'Внутри одного месяца - только движения'

       ELSE IF @НачалоПериода = DATESTART('MONTH', @НачалоПериода) THEN

              SET @ВариантОтчёта = 3 PRINT 'Начало месяца + движения конца'

            ELSE IF @КонецПериода = DATEEND('MONTH', @КонецПериода) THEN

                   SET @ВариантОтчёта = 4 PRINT 'Движения начала + конец месяца'

                 ELSE IF @НачалоИтогов = @КонецИтогов THEN

                        SET @ВариантОтчёта = 2 PRINT 'Через границу одного месяца'
                      
                      ELSE

                        SET @ВариантОтчёта = 5 PRINT 'Через границу больше одного месяца'

                      END
                 END
            END
       END
  END

  IF @ВариантОтчёта = 1 THEN
  
    SELECT Номенклатура, SUM(Количество) AS КоличествоОборот
      INTO @Результат
      FROM РегистрНакопления.ПродажиТовара.Итоги
     WHERE Период >= @НачалоПериода AND Период < @КонецПериода
     GROUP BY Номенклатура
    HAVING SUM(Количество) <> 0.0

  END -- # 1

  IF @ВариантОтчёта = 2 THEN
  
    SELECT Номенклатура, SUM(Количество) AS КоличествоОборот
      INTO @Результат
      FROM РегистрНакопления.ПродажиТовара
     WHERE Активность = TRUE AND Период >= @НачалоПериода AND Период <= @КонецПериода
     GROUP BY Номенклатура
    HAVING SUM(Количество) <> 0.0

  END -- # 2

  IF @ВариантОтчёта = 3 THEN

    WITH ИтогиПлюсДвижения AS
    (
      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара.Итоги
       WHERE Период >= @НачалоИтогов AND Период < @КонецИтогов
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0

      UNION ALL

      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара
       WHERE Активность = TRUE AND Период >= @КонецИтогов AND Период <= @КонецПериода
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0
    )
    SELECT Номенклатура, КоличествоОборот = SUM(Количество)
      INTO @Результат
      FROM ИтогиПлюсДвижения
     GROUP BY Номенклатура

  END -- # 3

  IF @ВариантОтчёта = 4 THEN

    WITH ИтогиПлюсДвижения AS
    (
      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара.Итоги
       WHERE Период >= @НачалоИтогов AND Период < @КонецИтогов
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0

      UNION ALL

      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара
       WHERE Активность = TRUE AND Период >= @НачалоПериода AND Период < @НачалоИтогов
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0
    )
    SELECT Номенклатура, КоличествоОборот = SUM(Количество)
      INTO @Результат
      FROM ИтогиПлюсДвижения
     GROUP BY Номенклатура

  END -- # 4

  IF @ВариантОтчёта = 5 THEN

    WITH ИтогиПлюсДвижения AS
    (
      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара.Итоги
       WHERE Период >= @НачалоИтогов AND Период < @КонецИтогов
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0
      
      UNION ALL

      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара
       WHERE Активность = TRUE AND Период >= @НачалоПериода AND Период < @НачалоИтогов
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0

      UNION ALL

      SELECT Номенклатура, SUM(Количество) AS Количество
        FROM РегистрНакопления.ПродажиТовара
       WHERE Активность = TRUE AND Период >= @КонецИтогов AND Период <= @КонецПериода
       GROUP BY Номенклатура
      HAVING SUM(Количество) <> 0.0
    )
    SELECT Номенклатура, КоличествоОборот = SUM(Количество)
      INTO @Результат
      FROM ИтогиПлюсДвижения
     GROUP BY Номенклатура

  END -- # 5

END -- USE

RETURN @Результат
```
