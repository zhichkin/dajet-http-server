# Таблица итогов регистра накопления оборотов

В отличие от регистров накопления остатков, регистры накопления оборотов не имеют актуальных итогов, а также установленного периода рассчитанных итогов. Период рассчитанных итогов для обротных регистров устанавливается платформой автоматически при включении использования итогов. Минимальный и максимальный периоды хранимых итогов вычисляются по основной таблице движений регистра запросами следующего вида:

```SQL
SELECT MIN(Период) FROM РегистрНакопления.ПродажиТовара -- Минимальный период
SELECT MAX(Период) FROM РегистрНакопления.ПродажиТовара -- Максимальный период
```

Полученные таким образом значения, приводятся к началу месяца. Например, если основная таблица движений содержит записи с 15 января 2026 года по 15 мая 2026 года, то минимальный период хранимых итогов будет равен ```2026-01-01 00:00:00```, а максимальный - ```2026-05-01 00:00:00```.

Пересчёт итогов выполняется для каждого месяца, начиная с минимального периода хранимых итогов и заканчивая максимальным. Запрос пересчёта итогов выполняется для каждого месяца в отдельной транзакции СУБД, имеющей уровень изоляции READ COMMITTED, и выглядит следующим образом:

```SQL
-- Пересчёт итогов регистра накопления оборотов за январь месяц

DECLARE @НачалоПериода datetime = '2026-01-01 00:00:00'
DECLARE @КонецПериода  datetime = '2026-02-01 00:00:00'

DELETE РегистрНакполения.ПродажиТовара.Итоги WHERE Период = @НачалоПериода

INSERT РегистрНакполения.ПродажиТовара.Итоги (Период, Номенклатура, Количество, Splitter)
SELECT @НачалоПериода, Номенклатура, SUM(Количество), 0
  FROM РегистрНакполения.ПродажиТовара
 WHERE Активность = TRUE AND Период >= @НачалоПериода AND Период < @КонецПериода
 GROUP BY Номенклатура
```

> **Вывод:** обороты за месяц хранятся на начало этого месяца. 

Рассмотрим запросы, которые платформа генерирует для получения оборотов.

**Используется только таблица итогов**

Самый оптимальный запрос: начало периода отчёта совпадает с началом месяца, а конец периода отчёта с концом месяца.

```SQL
DECLARE @НачалоПериода datetime = '2026-03-01 00:00:00' -- Начало периода отчёта
DECLARE @КонецПериода  datetime = '2026-04-30 23:59:59' -- Конец периода отчёта

-- Платформа приводит конец периода отчёта к началу следующего месяца
SET @КонецПериода = DATEADD('SECOND', @КонецПериода)

SELECT Номенклатура, SUM(Количество) AS КоличествоОборот
  FROM РегистрНакполения.ПродажиТовара.Итоги
 WHERE Период >= @НачалоПериода AND Период < @КонецПериода
 GROUP BY Номенклатура
HAVING SUM(Количество) <> 0.0
```

**Используются таблицы итогов и движений**

## Получение остатков при помощи DaJet Script

```SQL
DECLARE @ПериодОтчёта datetime = '2026-03-15T00:00:01' -- Период отчёта, на который нужно получить остатки

PRIVATE @ПериодИтогов     datetime = '0001-01-01T00:00:00' -- Период хранимых итогов для данного периода отчёта
PRIVATE @ПустойПериод     datetime = '0001-01-01T00:00:00' -- Значение пустой даты (константа)
PRIVATE @АктуальныйПериод datetime = '3999-11-01T00:00:00' -- Период актуальных итогов (константа)

PRIVATE @Результат array
PRIVATE @Настройки object

SET @ПериодИтогов = DATESTART('MONTH', @ПериодОтчёта) -- Значение по умолчанию: начало месяца периода отчёта

USE 'MS_TEST'

  SELECT UseTotals           -- numeric(1,0) Использовать итоги
       , ActualPeriod        -- boolean      Использовать актуальные итоги
       , MinCalculatedPeriod -- datetime     Минимальный период хранимых итогов
       , Period              -- datetime     Максимальный период хранимых итогов       
    INTO @Настройки
    FROM РегистрНакопления.ОстаткиТовара.Настройки

  -- Итоги не используются, период хранимых итогов не установлен и не используются актуальные итоги
  IF @Настройки.UseTotals = 1 OR (@Настройки.UseTotals = 0 AND @Настройки.MinCalculatedPeriod = @ПустойПериод AND @Настройки.ActualPeriod = FALSE) THEN

     SELECT Номенклатура
          , Остаток = SUM(CASE WHEN ВидДвижения = 0.0 THEN Количество ELSE -Количество END)
       INTO @Результат
       FROM РегистрНакопления.ОстаткиТовара
      WHERE Активность = TRUE AND Период < @ПериодОтчёта
      GROUP BY Номенклатура
     HAVING SUM(CASE WHEN ВидДвижения = 0.0 THEN Количество ELSE -Количество END) <> 0.0

     PRINT 'Итоги не используются, период хранимых итогов не установлен и не используются актуальные итоги'

     RETURN @Результат
  END

  -- Использование итогов включено, период хранимых итогов установлен и период отчёта совпадает с периодом хранимых итогов
  IF @ПериодОтчёта >= @Настройки.MinCalculatedPeriod AND @ПериодОтчёта <= @Настройки.Period AND @ПериодОтчёта = @ПериодИтогов THEN

    SELECT Номенклатура, Остаток = SUM(Количество)
      INTO @Результат
      FROM РегистрНакопления.ОстаткиТовара.Итоги
     WHERE Период = @ПериодИтогов
     GROUP BY Номенклатура
    HAVING SUM(Количество) <> 0.0

    PRINT 'Использование итогов включено, период хранимых итогов установлен и период отчёта совпадает с периодом хранимых итогов'

    RETURN @Результат
  END

  -- Использование итогов включено, период хранимых итогов установлен, актуальные итоги не используются
  IF @Настройки.Period > @ПустойПериод AND @Настройки.ActualPeriod = FALSE AND @ПериодОтчёта > @Настройки.Period THEN

     SET @ПериодИтогов = @Настройки.Period

     SELECT Номенклатура = ИтогиИДвижения.Номенклатура
          , Остаток      = SUM(ИтогиИДвижения.Количество)
       INTO @Результат
       FROM (SELECT Номенклатура = Итоги.Номенклатура
                  , Количество   = SUM(Итоги.Количество)
               FROM РегистрНакопления.ОстаткиТовара.Итоги AS Итоги
              WHERE Период = @ПериодИтогов
              GROUP BY Итоги.Номенклатура
             HAVING SUM(Итоги.Количество) <> 0.0
             UNION ALL
             SELECT Номенклатура = Движения.Номенклатура
                  , Количество   = SUM(CASE WHEN Движения.ВидДвижения = 0.0
                                            THEN  Движения.Количество
                                            ELSE -Движения.Количество END)
               FROM РегистрНакопления.ОстаткиТовара AS Движения
              WHERE Движения.Активность = TRUE
                AND Движения.Период >= @ПериодИтогов
                AND Движения.Период < @ПериодОтчёта
              GROUP BY Движения.Номенклатура
             HAVING SUM(CASE WHEN Движения.ВидДвижения = 0.0
                             THEN  Движения.Количество
                             ELSE -Движения.Количество END) <> 0.0) AS ИтогиИДвижения
      GROUP BY ИтогиИДвижения.Номенклатура
     HAVING SUM(ИтогиИДвижения.Количество) <> 0.0

     PRINT 'Использование итогов включено, период хранимых итогов установлен, актуальные итоги не используются'

     RETURN @Результат
  END

  -- Использование итогов включено, период хранимых итогов не установлен, но используются актуальные итоги
  IF @Настройки.MinCalculatedPeriod = @ПустойПериод AND @Настройки.ActualPeriod = TRUE THEN
     SET @ПериодИтогов = @АктуальныйПериод
  END

  -- Использование итогов включено, установлен период хранимых итогов,
  -- период отчёта больше максимального периода хранимых итогов,
  -- использование актуальных итогов включено
  IF @ПериодОтчёта > @Настройки.Period AND @Настройки.ActualPeriod = TRUE THEN
     SET @ПериодИтогов = @АктуальныйПериод
  END

  -- Использование итогов включено, установлен период хранимых итогов,
  -- период отчёта меньше минимального периода хранимых итогов
  IF @ПериодОтчёта < @Настройки.MinCalculatedPeriod THEN
     SET @ПериодИтогов = @Настройки.MinCalculatedPeriod
  END

  -- Использование итогов включено, установлен период хранимых итогов,
  -- период отчёта находится в диапазоне установленного периода хранимых итогов
  IF @ПериодОтчёта > @Настройки.MinCalculatedPeriod AND @ПериодОтчёта < @Настройки.Period THEN
     SET @ПериодИтогов = DATEADD('MONTH', 1, DATESTART('MONTH', @ПериодОтчёта))
  END

  PRINT 'Период итогов: ' + @ПериодИтогов
  PRINT 'Период отчёта: ' + @ПериодОтчёта
  PRINT 'Алгоритм вычитания движений из будущего по отношению к периоду отчёта итога'

  WITH ИтогиИДвижения AS
  (
     SELECT Номенклатура = Итоги.Номенклатура
          , Количество   = SUM(Итоги.Количество)
       FROM РегистрНакопления.ОстаткиТовара.Итоги AS Итоги
      WHERE Период = @ПериодИтогов
      GROUP BY Итоги.Номенклатура
     HAVING SUM(Итоги.Количество) <> 0.0
     
     UNION ALL
     
     SELECT Номенклатура = Движения.Номенклатура
          , Количество   = SUM(CASE WHEN Движения.ВидДвижения = 0.0
                                    THEN -Движения.Количество
                                    ELSE  Движения.Количество END)
       FROM РегистрНакопления.ОстаткиТовара AS Движения
      WHERE Движения.Активность = TRUE
        AND Движения.Период >= @ПериодОтчёта
        AND Движения.Период  < @ПериодИтогов
      GROUP BY Движения.Номенклатура
     HAVING SUM(CASE WHEN Движения.ВидДвижения = 0.0
                     THEN -Движения.Количество
                     ELSE  Движения.Количество END) <> 0.0
  )
  SELECT Номенклатура = ИтогиИДвижения.Номенклатура
       , Остаток      = SUM(ИтогиИДвижения.Количество)
    INTO @Результат
    FROM ИтогиИДвижения
   GROUP BY ИтогиИДвижения.Номенклатура
  HAVING SUM(ИтогиИДвижения.Количество) <> 0.0

  RETURN @Результат

END -- USE
```
